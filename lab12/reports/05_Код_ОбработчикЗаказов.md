# Код модуля ОбработчикЗаказов

```bsl
Процедура ДобавитьСобытие(НаборЗаписей, IDЗаказа, ТипСобытия, Данные, ВремяВыполнения)
    Запись = НаборЗаписей.Добавить();
    Запись.ID = Строка(IDЗаказа) + "|" + ТипСобытия;
    Запись.Тип = ТипСобытия;
    Запись.Данные = Данные;
    Запись.ВремяВыполнения = ВремяВыполнения;
    Запись.Статус = "Ожидает";
КонецПроцедуры

Процедура ПоставитьВОчередь(IDЗаказа) Экспорт
    НаборЗаписей = РегистрыСведений.ОчередьСобытий.СоздатьНаборЗаписей();
    ДобавитьСобытие(НаборЗаписей, IDЗаказа, "order:processing", "Заказ начал обрабатываться", ТекущаяДата() + 2);
    ДобавитьСобытие(НаборЗаписей, IDЗаказа, "order:complete", "Заказ обработан", ТекущаяДата() + 4);
    НаборЗаписей.Записать();
КонецПроцедуры

Процедура ОбработатьОчередь() Экспорт
    НаборЗаписей = РегистрыСведений.ОчередьСобытий.СоздатьНаборЗаписей();
    НаборЗаписей.Прочитать();

    Для Каждого Запись Из НаборЗаписей Цикл
        Если Запись.ВремяВыполнения <= ТекущаяДата()
            И Запись.Статус <> "Выполнено" Тогда

            ПозицияРазделителя = Найти(Запись.ID, "|");
            Если ПозицияРазделителя > 0 Тогда
                IDЗаказа = Лев(Запись.ID, ПозицияРазделителя - 1);
            Иначе
                IDЗаказа = Запись.ID;
            КонецЕсли;

            ДанныеСобытия = Новый Структура;
            ДанныеСобытия.Вставить("IDЗаказа", IDЗаказа);
            ДанныеСобытия.Вставить("Данные", Запись.Данные);
            СобытияСервера.ВызватьСобытие(Запись.Тип, ДанныеСобытия);
            Запись.Статус = "Выполнено";
        КонецЕсли;
    КонецЦикла;

    НаборЗаписей.Записать();
КонецПроцедуры
```
