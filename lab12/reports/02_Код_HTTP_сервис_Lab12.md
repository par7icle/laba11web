# Код HTTP-сервиса Lab12

```bsl
Функция RootGET(Запрос)
    Данные = Новый Структура;
    Данные.Вставить("method", "GET");
    Данные.Вставить("url", "/");
    СобытияСервера.ВызватьСобытие("request:received", Данные);
    Возврат ТекстовыйОтвет("Hello from Event-Driven Server!");
КонецФункции

Функция StartGET(Запрос)
    СобытияСервера.ВызватьСобытие("server:started", "Сервер запущен");
    Возврат ТекстовыйОтвет("Event server started");
КонецФункции

Функция StopGET(Запрос)
    СобытияСервера.ВызватьСобытие("server:stopped", "Сервер остановлен");
    Возврат ТекстовыйОтвет("Event server stopped");
КонецФункции

Функция OrderGET(Запрос)
    IDЗаказа = Запрос.ПараметрыURL["id"];
    СобытияСервера.ВызватьСобытие("order:start", "Заказ #" + IDЗаказа + " начал обработку");
    ОбработчикЗаказов.ПоставитьВОчередь(IDЗаказа);
    Возврат ТекстовыйОтвет("Заказ #" + IDЗаказа + " принят в обработку");
КонецФункции

Функция EventLoopGET(Запрос)
    EventLoopDemo.ЗапуститьДемонстрацию();
    Возврат ТекстовыйОтвет("Демонстрация Event Loop выполнена");
КонецФункции

Функция UserActionGET(Запрос)
    UserId = Запрос.ПараметрыЗапроса.Получить("userId");
    Action = Запрос.ПараметрыЗапроса.Получить("action");
    Если UserId = Неопределено Тогда UserId = ""; КонецЕсли;
    Если Action = Неопределено Тогда Action = ""; КонецЕсли;
    Событие = Новый Структура;
    Событие.Вставить("userId", UserId);
    Событие.Вставить("action", Action);
    Событие.Вставить("timestamp", ТекущаяДата());
    Событие.Вставить("metadata", "HTTP request");
    Событие.Вставить("id", Строка(Новый УникальныйИдентификатор));
    СобытияСервера.ВызватьСобытие("user:action", Событие);
    Возврат ТекстовыйОтвет("user:action обработан");
КонецФункции

Функция ТекстовыйОтвет(Текст)
    Ответ = Новый HTTPСервисОтвет(200);
    Ответ.Заголовки.Вставить("Content-Type", "text/plain; charset=utf-8");
    Ответ.УстановитьТелоИзСтроки(Текст, КодировкаТекста.UTF8);
    Возврат Ответ;
КонецФункции
```
