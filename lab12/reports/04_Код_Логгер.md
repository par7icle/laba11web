# Код модуля Логгер

```bsl
Процедура Записать(Событие, Данные = Неопределено) Экспорт

    ТекстДанных = "";

    Если Данные = Неопределено Тогда
        ТекстДанных = "";
    ИначеЕсли ТипЗнч(Данные) = Тип("Структура") Тогда
        Для Каждого Элемент Из Данные Цикл
            Если ТекстДанных <> "" Тогда
                ТекстДанных = ТекстДанных + "; ";
            КонецЕсли;
            ТекстДанных = ТекстДанных + Строка(Элемент.Ключ) + "=" + Строка(Элемент.Значение);
        КонецЦикла;
    Иначе
        ТекстДанных = Строка(Данные);
    КонецЕсли;

    Сообщение = "[" + Строка(ТекущаяДата()) + "] " + Событие;
    Если ТекстДанных <> "" Тогда
        Сообщение = Сообщение + ": " + ТекстДанных;
    КонецЕсли;

    ЗаписьЖурналаРегистрации("Lab12", УровеньЖурналаРегистрации.Информация, , , Сообщение);

КонецПроцедуры
```
